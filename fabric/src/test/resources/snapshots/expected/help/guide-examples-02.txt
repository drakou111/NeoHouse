long p.score;
long g.highScore = 0;

function addScore(long n) {
    p.score += n;
    if (p.score > g.highScore) {
        g.highScore = p.score;
    }
    if (p.score >= 1000) {
        message("You won with " + p.score + " points!");
        p.score = 0;
    }
}

command score(listed: true) {
    message("Score: " + p.score + " | Best: " + g.highScore);
}

event onKill { addScore(10); }
