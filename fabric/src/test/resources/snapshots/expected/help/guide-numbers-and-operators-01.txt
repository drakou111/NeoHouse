event onJoin {
    p.speed = 2.7;
    p.whole = (long) p.speed;              // 2
    p.total = p.speed + (double) p.whole;  // both sides are doubles
    p.label = (string) p.whole;            // "2"
    if ((long) p.speed > 1) {
        message("Speed: " + p.speed);      // no cast needed when joining text
    }
}
