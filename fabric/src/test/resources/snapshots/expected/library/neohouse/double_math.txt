// Working variables of this file, declared so scripts that require declarations can import it.
double p.doubleTemp;

// Returns the absolute value of x.
function double double_abs(double x) {
    if (x < 0.0) { return -x; }
    return x;
}

// Returns the smaller of a and b.
function double double_min(double a, double b) {
    if (a < b) { return a; }
    return b;
}

// Returns the larger of a and b.
function double double_max(double a, double b) {
    if (a > b) { return a; }
    return b;
}

// Clamps x to the range [lo, hi].
function double double_clamp(double x, double lo, double hi) {
    if (x < lo) { return lo; }
    if (x > hi) { return hi; }
    return x;
}

// Sign of x: returns -1.0, 0.0, or 1.0.
function double double_sign(double x) {
    if (x < 0.0) { return -1.0; }
    if (x > 0.0) { return 1.0; }
    return 0.0;
}

// Remainder of x divided by n.
function double double_mod(double x, double n) {
    p.doubleTemp = (double)((long)(x / n));
    p.doubleTemp = x - n * p.doubleTemp;

    if (p.doubleTemp < 0.0) {
        p.doubleTemp += n;
    }

    return (double)p.doubleTemp;
}

// Square root of x.
function double sqrt(double x) {
    p.doubleTemp = 0.001;
    if (x <= 0.0) { p.doubleTemp = 0.0; }
    if (x >= 0.03) { p.doubleTemp = 0.5; }
    if (x >= 0.001 && x < 0.03) { p.doubleTemp = 0.1; }
    if (x > 0.0) {
        repeat(8) {
            p.doubleTemp = (p.doubleTemp + x / p.doubleTemp) * 0.5;
        }
    }
    else {
        p.doubleTemp = 0.0;
    }
    return (double)p.doubleTemp;
}
