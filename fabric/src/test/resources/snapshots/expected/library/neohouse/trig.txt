// Working variables of this file, declared so scripts that require declarations can import it.
double p.trigTemp;
double p.trigTemp2;
double p.trigTemp3;
double p.trigTemp4;
double p.trigTemp5;
double p.trigYaw;
double p.trigPitch;
double p.trigCosYaw;
double p.trigCosPitch;
double p.trigSinYaw;
double p.trigSinPitch;

// We use Bhāskara I's approximation, which is extremely precise for values [0,π].
// sin(x) ≈ B(x) = (16x(π-x)) / (5π^2-4x(π-x)).
// To extend to ]-∞, ∞[, we can do this:
// y = mod(x, 2π)
// t = π - abs(π - y)
// sin(x) ≈ sign(π - y) * B(t)
function double sin(double x) {
    // y = mod(x, 2π)
    p.trigTemp = (double)((long)(x / TWO_PI));
    p.trigTemp = x - TWO_PI * p.trigTemp;
    if (p.trigTemp < 0.0) {
        p.trigTemp += TWO_PI;
    }

    // t = π - abs(π - y)
    p.trigTemp2 = PI - p.trigTemp;
    if (p.trigTemp2 < 0.0) { p.trigTemp2 *= -1.0; }
    p.trigTemp2 = PI - p.trigTemp2;

    // B(t)
    p.trigTemp3 = (16.0 * p.trigTemp2 * (PI - p.trigTemp2)) / (49.3480220054 - 4.0 * p.trigTemp2 * (PI - p.trigTemp2));

    // sign(π - y)
    p.trigTemp4 = 0.0;
    if (p.trigTemp < PI) { p.trigTemp4 = 1.0; }
    if (p.trigTemp > PI) { p.trigTemp4 = -1.0; }

    // sin(x) ≈ sign(π - y) * B(t)
    return (double)(p.trigTemp4 * p.trigTemp3);
}

// cos(x) = sin(x + π/2), written out instead of calling sin: Housing won't run a function
// twice in one execution, and tan needs both sin and cos.
function double cos(double x) {
    p.trigTemp5 = x + 1.57079632679;

    // y = mod(x + π/2, 2π)
    p.trigTemp = (double)((long)(p.trigTemp5 / TWO_PI));
    p.trigTemp = p.trigTemp5 - TWO_PI * p.trigTemp;
    if (p.trigTemp < 0.0) {
        p.trigTemp += TWO_PI;
    }

    // t = π - abs(π - y)
    p.trigTemp2 = PI - p.trigTemp;
    if (p.trigTemp2 < 0.0) { p.trigTemp2 *= -1.0; }
    p.trigTemp2 = PI - p.trigTemp2;

    // B(t)
    p.trigTemp3 = (16.0 * p.trigTemp2 * (PI - p.trigTemp2)) / (49.3480220054 - 4.0 * p.trigTemp2 * (PI - p.trigTemp2));

    // sign(π - y)
    p.trigTemp4 = 0.0;
    if (p.trigTemp < PI) { p.trigTemp4 = 1.0; }
    if (p.trigTemp > PI) { p.trigTemp4 = -1.0; }

    return (double)(p.trigTemp4 * p.trigTemp3);
}

// tan(x) = sin(x)/cos(x).
function double tan(double x) {
    return sin(x)/cos(x);
}

// let u = sqrt((1-|x|)/2)
// asin(x) ≈ sign(x) * (π/2 - 2*u*(
//   0.9999994098 + 0.1667511541*u^2 + 0.07308794137*u^4 +
//   0.05996102211*u^6 - 0.0220605251*u^8 + 0.09460451869*u^10
// ))
function double asin(double x) {
    // (1-|x|)/2
    p.trigTemp = x;
    if (p.trigTemp < 0.0) { p.trigTemp *= -1.0; }
    p.trigTemp = (1.0 - p.trigTemp) / 2.0;

    // u = sqrt((1-|x|)/2)
    p.trigTemp2 = 0.0316;
    if (p.trigTemp <= 0.0) { p.trigTemp2 = 0.0; }
    if (p.trigTemp >= 0.03) { p.trigTemp2 = 0.5; }
    if (p.trigTemp >= 0.001 && p.trigTemp < 0.03) { p.trigTemp2 = 0.1; }
    if (p.trigTemp > 0.0) {
        repeat(4) {
            p.trigTemp2 += p.trigTemp / p.trigTemp2;
            p.trigTemp2 *= 0.5;
        }
    }
    else {
        p.trigTemp2 = 0.0;
    }

    // The polynomial
    p.trigTemp3 = p.trigTemp2 * p.trigTemp2;
    p.trigTemp = 0.09460451869;
    p.trigTemp = p.trigTemp * p.trigTemp3 - 0.0220605251;
    p.trigTemp = p.trigTemp * p.trigTemp3 + 0.05996102211;
    p.trigTemp = p.trigTemp * p.trigTemp3 + 0.07308794137;
    p.trigTemp = p.trigTemp * p.trigTemp3 + 0.1667511541;
    p.trigTemp = p.trigTemp * p.trigTemp3 + 0.9999994098;

    // asin(x) ≈ sign(x) * (π/2 - 2*u*(...))
    p.trigTemp = 1.57079632679 - 2.0 * p.trigTemp2 * p.trigTemp;
    if (x < 0.0) { p.trigTemp *= -1.0; }

    return (double)p.trigTemp;
}

// acos(x) = π/2 - asin(x).
function double acos(double x) {
    return 1.57079632679 - asin(x);
}

// atan(x) = asin(x / sqrt(1 + x^2))
function double atan(double x) {
    // 1 + x^2
    p.trigTemp = 1.0 + x * x;

    // sqrt(1 + x^2)
    p.trigTemp2 = 0.0316;
    if (p.trigTemp <= 0.0) { p.trigTemp2 = 0.0; }
    if (p.trigTemp >= 0.03) { p.trigTemp2 = 0.5; }
    if (p.trigTemp >= 0.001 && p.trigTemp < 0.03) { p.trigTemp2 = 0.1; }
    if (p.trigTemp > 0.0) {
        repeat(4) {
            p.trigTemp2 += p.trigTemp / p.trigTemp2;
            p.trigTemp2 *= 0.5;
        }
    }
    else {
        p.trigTemp2 = 0.0;
    }

    // atan(x) = asin(x / sqrt(1 + x^2))
    return asin(x / p.trigTemp2);
}

function double atan2(double x, double y) {
    if (x > 0.0)                { return atan(y / x); }
    if (x < 0.0 && y >= 0.0)    { return atan(y / x) + 3.14159265359; }
    if (x < 0.0 && y < 0.0)     { return atan(y / x) - 3.14159265359; }
    if (y > 0.0)                { return  1.57079632679; }
    if (y < 0.0)                { return -1.57079632679; }
    return 0.0;
}

// The unit vector the player is looking along, as (x, y, z), from their yaw and pitch.
function (double, double, double) getFacingVector() {
    p.trigYaw = 0.01745329251 * ph.playerPosYaw();
    p.trigPitch = 0.01745329251 * ph.playerPosPitch();
    p.trigCosYaw = cos(p.trigYaw);
    p.trigSinYaw = sin(p.trigYaw);

    // sin and cos can't be called a second time here (Housing won't run a function twice in one
    // execution). Pitch is within [-pi/2, pi/2], so both reduce to B(t) on [0, pi] directly.
    // cos(pitch) = B(pitch + pi/2)
    p.trigTemp2 = p.trigPitch + 1.57079632679;
    p.trigCosPitch = (16.0 * p.trigTemp2 * (PI - p.trigTemp2)) / (49.3480220054 - 4.0 * p.trigTemp2 * (PI - p.trigTemp2));

    // sin(pitch) = sign(pitch) * B(|pitch|)
    p.trigTemp2 = p.trigPitch;
    if (p.trigTemp2 < 0.0) { p.trigTemp2 *= -1.0; }
    p.trigSinPitch = (16.0 * p.trigTemp2 * (PI - p.trigTemp2)) / (49.3480220054 - 4.0 * p.trigTemp2 * (PI - p.trigTemp2));
    if (p.trigPitch < 0.0) { p.trigSinPitch *= -1.0; }

    return ((double)(p.trigSinYaw * p.trigCosPitch * -1.0), (double)(p.trigSinPitch * -1.0), (double)(p.trigCosYaw * p.trigCosPitch));
}
